bigshrimp/
│
├── main.py                      # 程序总入口：初始化应用、加载全局样式、启动总控
├── requirements.txt             # 依赖声明（PySide2, websockets, pymysql, pymssql 等）
│
├── model/                       # 【Model 层】数据、状态与配置持久化
│   ├── __init__.py
│   ├── project_model.py         # 项目模型：管理工作区路径、文件树缓存、当前打开的 Tab 状态
│   ├── config_model.py          # 配置模型：基于 QSettings 持久化窗口大小、历史参数、最近项目
│   └── agent_model.py           # Agent 模型：管理大模型会话上下文、Token 消耗、提示词模板
│
├── view/                        # 【View 层】纯界面展示与哑视图布局（零业务逻辑）
│   ├── __init__.py
│   ├── main_window.py           # 主窗口：顶层菜单栏、状态栏、主多 Tab 容器、全局分割器
│   ├── project_tree_view.py     # 左侧面板：文件目录树渲染、右键菜单视觉触发
│   ├── runner_view.py           # 沙箱控制面板：运行/停止按钮、执行参数输入框
│   ├── console_view.py          # 终端控制台：富文本日志流、清屏、输出过滤展示
│   ├── source_editor_view.py    # 源码编辑器：代码高亮、行号侧边栏、Diff 对比与大纲面板
│   └── agent_panel_view.py      # AI 交互面板：多场景 Prompt 模板多 Tab、对话流、高危操作审批流
│
├── controller/                  # 【Controller 层】业务调度、事件响应与状态机枢纽
│   ├── __init__.py
│   ├── main_controller.py       # 总控制器：生命周期管理、依赖注入、跨模块总线协调
│   ├── project_controller.py    # 项目控制器：目录打开、文件导航、树状事件响应
│   ├── runner_controller.py     # 运行控制器：沙箱进程启停、控制台日志同步、自愈合闭环触发
│   ├── editor_controller.py     # 编辑控制器：源码加载、文件保存、静态大纲联动解析
│   └── agent_controller.py      # Agent 控制器：编排 WebSocket 桥接、处理 AI 响应、自动落盘调度
│
├── service/                     # 【Service 层】独立底层服务（解耦 UI，支持独立单元测试）
│   ├── __init__.py
│   ├── scanner_service.py       # 扫描服务：AST 语法树解析、识别脚本主入口
│   ├── executor_service.py      # 执行服务：封装 QProcess 异步安全运行 Python 沙箱子进程
│   ├── analyzer_service.py      # 分析服务：提取类/函数大纲符号、统计代码行数
│   ├── bridge_service.py        # 桥接服务：本地 QWebSocketServer，与浏览器插件双向通信
│   ├── openclaw_gateway_service.py # 网关服务：适配外部通道协议（支持移动端或跨平台远程指令接入）
│   ├── db_inspector_service.py  # 数据库感知服务：深度探测 MySQL/SQL Server Schema 供 Agent 注入
│   └── self_heal_service.py     # 自愈合服务：拦截沙箱 Traceback 报错，自动构建修复 Prompt 闭环
│
└── resources/                   # 【资源层】静态资产
    ├── icons/                   # 系统 UI 图标资产
    └── qss/
        └── dark_style.qss       # 全局现代暗黑主题样式表
